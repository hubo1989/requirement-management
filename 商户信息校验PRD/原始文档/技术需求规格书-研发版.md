```sql
-- 校验批次表
CREATE TABLE validation_batch (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    batch_no VARCHAR(20) UNIQUE NOT NULL,
    file_name VARCHAR(255) NOT NULL,
    file_path VARCHAR(500) NOT NULL,
    file_size BIGINT NOT NULL,
    operator VARCHAR(50) NOT NULL,
    institution_bank VARCHAR(100),
    status ENUM('success', 'error', 'pending') DEFAULT 'pending',
    total_count INT DEFAULT 0,
    success_count INT DEFAULT 0,
    error_count INT DEFAULT 0,
    pending_count INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_batch_no (batch_no),
    INDEX idx_operator (operator),
    INDEX idx_status (status),
    INDEX idx_created_at (created_at)
);

-- 校验详情表
CREATE TABLE validation_detail (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    batch_id BIGINT NOT NULL,
    row_number INT NOT NULL,
    data_json JSON NOT NULL,
    validation_result JSON NOT NULL,
    status ENUM('valid', 'invalid', 'pending') DEFAULT 'pending',
    error_message TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (batch_id) REFERENCES validation_batch(id),
    INDEX idx_batch_id (batch_id),
    INDEX idx_status (status)
);

-- 操作日志表
CREATE TABLE operation_log (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id VARCHAR(50) NOT NULL,
    operation_type VARCHAR(50) NOT NULL,
    operation_desc TEXT,
    request_params JSON,
    response_result JSON,
    ip_address VARCHAR(50),
    user_agent TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_user_id (user_id),
    INDEX idx_operation_type (operation_type),
    INDEX idx_created_at (created_at)
);
```

### 校验规则引擎
```java
@Component
public class ValidationEngine {
    
    private final List<Validator> validators = Arrays.asList(
        new PhoneNumberValidator(),
        new IdCardValidator(),
        new EmailValidator(),
        new BankCardValidator()
    );
    
    public ValidationResult validate(Map<String, Object> data) {
        ValidationResult result = new ValidationResult();
        
        for (Validator validator : validators) {
            ValidationResultItem item = validator.validate(data);
            result.addItem(item);
        }
        
        return result;
    }
}
```

## 🔒 安全设计

### 认证授权
```java
@Configuration
@EnableWebSecurity
public class SecurityConfig {
    
    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            .authorizeHttpRequests(authz -> authz
                .requestMatchers("/api/upload/**").hasRole("OPERATOR")
                .requestMatchers("/api/history/**").hasRole("ADMIN")
                .anyRequest().authenticated()
            )
            .oauth2Login(withDefaults());
        
        return http.build();
    }
}
```

### 应用监控
```java
@Component
public class MetricsCollector {
    
    private final MeterRegistry meterRegistry;
    private final Counter fileUploadCounter;
    private final Timer validationTimer;
    
    public MetricsCollector(MeterRegistry meterRegistry) {
        this.meterRegistry = meterRegistry;
        this.fileUploadCounter = Counter.builder("file.upload.count").register(meterRegistry);
        this.validationTimer = Timer.builder("validation.time").register(meterRegistry);
    }
    
    public void recordFileUpload() {
        fileUploadCounter.increment();
    }
    
    public void recordValidationTime(Duration duration) {
        validationTimer.record(duration);
    }
}
```
